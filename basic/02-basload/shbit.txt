REM
REM BASLOAD - BASIC WITHOUT LINE NUMBERS
REM ON THE COMMANDER X16 SIMPLY USE
REM BASLOAD "SHBIT.TXT"
REM TO LOAD THIS
REM
REM IT WILL CREATE LINE NUMBERS, VARIABLES AND
REM MAKE SURE THE GOTO'S USED ARE WORKING.
REM
REM 8 BIT VARIABLE
REM
REM /=v=v=v=v=v=v=v=\
REM |7|6|5|4|3|2|1|0|
REM +=x=x=x=x=x=x=x=+
REM |7|6|5|4|3|2|1|0|
REM \=^=^=^=^=^=^=^=/
REM

VAR.V=42: REM INPUT VAR

BOX.I=0

REM BOX - TOP PART
PRINT CHR$($B0);
FOR BOX.I=0 TO 7-1
: PRINT CHR$($63);CHR$($B2);
NEXT BOX.I
PRINT CHR$($63);CHR$($AE);
PRINT

REM BOX - LEGEND
FOR BOX.I=0 TO 7
: PRINT CHR$($62);MID$(STR$(7-BOX.I),2);
NEXT BOX.I
PRINT CHR$($62);
PRINT

REM BOX - MIDDLE PART
PRINT CHR$($AB);
FOR BOX.I=0 TO 7-1
: PRINT CHR$($63);CHR$($7B);
NEXT BOX.I
PRINT CHR$($63);CHR$(179);
PRINT

REM BOX - VARIABLE PART
FOR BOX.I=7 TO 0 STEP -1
: IF (VAR.V AND 2^BOX.I) THEN PRINT CHR$($62);"1";:GOTO VARPART.SKIP
: PRINT CHR$($62);"0";
VARPART.SKIP:
NEXT BOX.I
PRINT CHR$($62);
PRINT

REM BOX - LOWER PART
PRINT CHR$(173);
FOR BOX.I=0 TO 7-1
: PRINT CHR$($63);CHR$(177);
NEXT BOX.I
PRINT CHR$($63);CHR$(189);
PRINT
